// Employee.java
class Employee {
    // Attributes of the Employee class
    String name;
    int id;
    double salary;

    // Constructor to initialize the Employee object
    Employee(String name, int id, double salary) {
        this.name = name;
        this.id = id;
        this.salary = salary;
    }

    // Method to display the details of the employee
    void displayDetails() {
        System.out.println("Employee Details:");
        System.out.println("Name: " + name);
        System.out.println("ID: " + id);
        System.out.println("Salary: $" + salary);
    }
}

// Main class to test the Employee class
public class Main {
    public static void main(String[] args) {
        // Create an Employee object
        Employee emp1 = new Employee("John Doe", 101, 50000.0);

        // Call the method to display employee details
        emp1.displayDetails();
    }
}
