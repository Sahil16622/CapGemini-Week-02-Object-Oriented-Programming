// MobilePhone.java
class MobilePhone {
    // Attributes of the MobilePhone class
    String brand;
    String model;
    double price;

    // Constructor to initialize the MobilePhone object
    MobilePhone(String brand, String model, double price) {
        this.brand = brand;
        this.model = model;
        this.price = price;
    }

    // Method to display the details of the mobile phone
    void displayDetails() {
        System.out.println("Mobile Phone Details:");
        System.out.println("Brand: " + brand);
        System.out.println("Model: " + model);
        System.out.println("Price: $" + price);
    }
}

// Main class to test the MobilePhone class
public class Main {
    public static void main(String[] args) {
        // Create a MobilePhone object
        MobilePhone phone1 = new MobilePhone("Samsung", "Galaxy S21", 799.99);

        // Call the method to display phone details
        phone1.displayDetails();
    }
}
